### b) Persistent Messages with Session Storage

Add session storage to `static/index.html`:
```html
<script>
  const ws = new WebSocket(`ws://${location.host}/ws`);
  const messages = document.getElementById('messages');
  const input = document.getElementById('input');

  // Load previous messages from sessionStorage
  let messageHistory = JSON.parse(sessionStorage.getItem('chatMessages') || '[]');
  messageHistory.forEach(msg => {
    const p = document.createElement('p');
    p.textContent = `MSG: ${msg}`;
    messages.appendChild(p);
  });

  input.onkeypress = e => {
    if (e.key === 'Enter' && input.value.trim()) {
      ws.send(input.value);
      input.value = '';
    }
  };

  ws.onmessage = ({data}) => {
    // Update storage and DOM
    messageHistory.push(data);
    sessionStorage.setItem('chatMessages', JSON.stringify(messageHistory));
    
    const p = document.createElement('p');
    p.textContent = `MSG: ${data}`;
    messages.appendChild(p);
    messages.scrollTop = messages.scrollHeight;
  };
</script>
```

This implementation will:
1. Read the HTML file fresh on each request for hot-reloading
2. Persist chat messages in session storage during the browser tab's lifetime
3. Automatically clear messages when the tab closes (using sessionStorage)
